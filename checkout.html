Pricing Engine --- const urlParams = new URLSearchParams(window.location.search); const projectParam = urlParams.get('project'); const priceParam = urlParams.get('price'); let numericPrice = priceParam ? parseFloat(priceParam) : 0; if (projectParam) document.getElementById('projectName').innerText = projectParam.replace(/-/g, ' '); if (priceParam) { let displayPrice = "₹" + numericPrice.toLocaleString('en-IN', { minimumFractionDigits: 2 }); document.getElementById('basePrice').innerText = displayPrice; document.getElementById('totalPrice').innerText = displayPrice; document.getElementById('btnPrice').innerText = displayPrice; } // --- 2. Unified Submit Action --- async function handleOrderSubmit(e) { e.preventDefault(); // This STOPS the page from refreshing const submitBtn = e.target.querySelector('button[type="submit"]'); submitBtn.innerHTML = "Processing..."; submitBtn.disabled = true; try { const backendURL = "/.netlify/functions/paytm"; const email = document.querySelector('input[name="email"]').value; const response = await fetch(backendURL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ amount: numericPrice, email: email }) }); if (!response.ok) throw new Error("Backend failed"); const data = await response.json(); // Paytm Popup Logic var config = { "root": "", "flow": "DEFAULT", "data": { "orderId": data.orderId, "token": data.token, "tokenType": "TXN_TOKEN", "amount": data.amount }, "handler": { "notifyMerchant": (e, d) => console.log(e, d) } }; if (window.Paytm && window.Paytm.CheckoutJS) { window.Paytm.CheckoutJS.init(config).then(() => window.Paytm.CheckoutJS.invoke()); } else { alert("Paytm Gateway not loaded."); } } catch (err) { console.error(err); alert("Server error. Check Netlify logs."); } finally { submitBtn.innerHTML = "Pay Securely"; submitBtn.disabled = false; } }
